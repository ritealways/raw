import os
from openai import OpenAI
from dotenv import load_dotenv

# Load environment variables from your .env file
load_dotenv()

def test_openai_connection():
    # Retrieve the API key
    api_key = os.getenv("OPENAI_API_KEY")
    
    if not api_key:
        print("❌ Error: OPENAI_API_KEY is missing from your environment variables or .env file.")
        return

    print("🔑 API Key detected. Attempting to connect to OpenAI...")
    
    try:
        # Initialize the OpenAI client
        client = OpenAI(api_key=api_key)
        
        # Send a minimal test request
        response = client.chat.completions.create(
            model="gpt-3.5-turbo",  # You can change this to your preferred model
            messages=[{"role": "user", "content": "Say 'Connected successfully!' if you can read this."}],
            max_tokens=15
        )
        
        print("\n✅ Connection Working Properly!")
        print("Response from model:", response.choices[0].message.content.strip())
        
    except Exception as e:
        print(f"\n❌ Connection Failed. Error details:\n{e}")

if __name__ == "__main__":
    test_openai_connection()
