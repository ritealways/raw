import numpy as np
import pandas as pd

# -----------------------------
# MAPE function
# -----------------------------
def mape(y_true, y_pred, eps=1e-6):
    y_true = np.asarray(y_true)
    y_pred = np.asarray(y_pred)

    return float(
        np.mean(
            np.abs(y_true - y_pred) /
            (np.abs(y_true) + eps)
        ) * 100
    )
