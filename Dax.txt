        // ============================================================
        //  DOM REFS
        // ============================================================
        const bodyNav = document.getElementById('bodyNav');
        const exerciseGrid = document.getElementById('exerciseGrid');
        const sectionTitle = document.getElementById('sectionTitle');
        const bodyNameSpan = document.getElementById('bodyName');
        const exerciseCountSpan = document.getElementById('exerciseCount');

        const modalOverlay = document.getElementById('modalOverlay');
        const modalContent = document.getElementById('modalContent');
        const modalImgSrc = document.getElementById('modalImgSrc');
        const modalTitle = document.getElementById('modalTitle');
        const modalFocus = document.getElementById('modalFocus');
        const modalTags = document.getElementById('modalTags');
        const modalSteps = document.getElementById('modalSteps');
        const modalCloseBtn = document.getElementById('modalCloseBtn');

        // ============================================================
        //  RENDER NAV
        // ============================================================
        function renderNav() {
            const keys = Object.keys(EXERCISES_DATA);
            bodyNav.innerHTML = keys.map(key => {
                const data = EXERCISES_DATA[key];
                const active = key === currentBody ? 'active' : '';
                return `
                    <button class="body-nav-btn ${active}" data-body="${key}">
                        <i class="fas ${data.icon}"></i>
                        ${data.label}
                    </button>
                `;
            }).join('');

            // event listeners
            bodyNav.querySelectorAll('.body-nav-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    const body = btn.dataset.body;
                    if (body && body !== currentBody) {
                        currentBody = body;
                        renderNav();
                        renderExercises();
                    }
                });
            });
        }

        // ============================================================
        //  RENDER EXERCISES
        // ============================================================
        function renderExercises() {
            const data = EXERCISES_DATA[currentBody];
            if (!data) return;

            bodyNameSpan.textContent = data.label;
            const exercises = data.exercises;
            exerciseCountSpan.textContent = `${exercises.length} exercises`;

            if (exercises.length === 0) {
                exerciseGrid.innerHTML = `
                    <div class="empty-state">
                        <i class="fas fa-dumbbell"></i>
                        <p>No exercises for this body part yet.</p>
                    </div>
                `;
                return;
            }

            exerciseGrid.innerHTML = exercises.map((ex, index) => `
                <div class="exercise-card" data-index="${index}" data-body="${currentBody}">
                    <div class="exercise-img-wrap">
                        <img src="${ex.image}" alt="${ex.name}" loading="lazy" />
                        <span class="exercise-badge"><i class="fas fa-play"></i> demo</span>
                    </div>
                    <div class="exercise-info">
                        <h3>${ex.name}</h3>
                        <div class="focus"><i class="fas fa-bullseye"></i> ${ex.focus}</div>
                        <div class="exercise-meta">
                            <span class="tag"><i class="fas fa-tag"></i> ${ex.equipment}</span>
                        </div>
                    </div>
                </div>
            `).join('');

            // attach click listeners for modal
            exerciseGrid.querySelectorAll('.exercise-card').forEach(card => {
                card.addEventListener('click', () => {
                    const idx = parseInt(card.dataset.index);
                    const bodyKey = card.dataset.body;
                    const exData = EXERCISES_DATA[bodyKey]?.exercises[idx];
                    if (exData) openModal(exData);
                });
            });
        }

        // ============================================================
        //  MODAL
        // ============================================================
        function openModal(ex) {
            modalImgSrc.src = ex.image;
            modalImgSrc.alt = ex.name;
            modalTitle.textContent = ex.name;
            modalFocus.textContent = ex.focus;

            // tags
            modalTags.innerHTML = `
                <span class="tag"><i class="fas fa-tag"></i> ${ex.equipment}</span>
                <span class="tag"><i class="fas fa-bullseye"></i> ${ex.focus}</span>
            `;

            // steps
            modalSteps.innerHTML = ex.steps.map((step, i) => `
                <li><span class="step-num">${i + 1}.</span> ${step}</li>
            `).join('');

            modalOverlay.classList.add('open');
            document.body.style.overflow = 'hidden';
        }

        function closeModal() {
            modalOverlay.classList.remove('open');
            document.body.style.overflow = '';
        }

        // close events
        modalCloseBtn.addEventListener('click', closeModal);
        modalOverlay.addEventListener('click', (e) => {
            if (e.target === modalOverlay) closeModal();
        });
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') closeModal();
        });

        // ============================================================
        //  INIT
        // ============================================================
        renderNav();
        renderExercises();

        // small console greeting
        console.log('🏋️ MuscleMap Pro loaded — built for Coach Alex.');
    </script>
