@st.cache_data
def load_premium_data():
    """Load the WC Premium Forecast dataset."""
    try:
        df = pd.read_excel(project_root / 'WC_Premium_Forecast_Dataset_Noisy.xlsx')
        if 'Month' in df.columns:
            df['Month'] = pd.to_datetime(df['Month'], errors='coerce')
        return df
    except Exception as e:
        st.warning(f"Could not load Premium file: {e}. Using sample data.")
        
        # Dynamic date range: past 3 years up to the current month
        end_date = pd.Timestamp.today().normalize().replace(day=1)
        start_date = end_date - pd.DateOffset(years=3)
        dates = pd.date_range(start=start_date, end=end_date, freq='MS')
        
        return pd.DataFrame({
            'Month': dates,
            'GWP': np.random.randint(50000, 200000, len(dates)),
            'Renewal_Uplift': np.random.uniform(0.05, 0.25, len(dates)),
            'Retention_Rate': np.random.uniform(0.70, 0.95, len(dates)),
            'Endorsement_Frequency': np.random.uniform(0.5, 2.5, len(dates)),
            'Policy_Term_Length': np.random.uniform(6, 24, len(dates)),
            'Loss_Ratio': np.random.uniform(0.6, 0.9, len(dates))
        })
